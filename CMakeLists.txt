cmake_minimum_required(VERSION 3.21)
project(SkyrimTutorials VERSION 0.4.1 LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "A Windows x64 toolchain is required")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(SkyrimTutorials SHARED src/Plugin.cpp src/Library.cpp src/Notes.cpp src/Browser.cpp src/DisplayText.cpp src/Documents.cpp)
target_link_libraries(SkyrimTutorials PRIVATE crypt32)
target_include_directories(SkyrimTutorials PRIVATE src vendor)
target_compile_features(SkyrimTutorials PRIVATE cxx_std_20)
target_compile_definitions(SkyrimTutorials PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
target_compile_options(SkyrimTutorials PRIVATE /W4 /EHsc /utf-8 /permissive-)
set_target_properties(SkyrimTutorials PROPERTIES PREFIX "")
add_executable(LibraryCheck tools/LibraryCheck.cpp src/Library.cpp src/DisplayText.cpp src/Documents.cpp)
target_include_directories(LibraryCheck PRIVATE src vendor)
target_compile_features(LibraryCheck PRIVATE cxx_std_20)
target_compile_definitions(LibraryCheck PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
target_compile_options(LibraryCheck PRIVATE /W4 /EHsc /utf-8)
add_executable(NotesCheck tools/NotesCheck.cpp src/Notes.cpp src/Library.cpp src/DisplayText.cpp src/Documents.cpp)
target_include_directories(NotesCheck PRIVATE src vendor)
target_compile_features(NotesCheck PRIVATE cxx_std_20)
target_compile_definitions(NotesCheck PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
target_compile_options(NotesCheck PRIVATE /W4 /EHsc /utf-8)
target_link_libraries(NotesCheck PRIVATE crypt32)

add_executable(BrowserCheck tools/BrowserCheck.cpp src/Library.cpp src/Browser.cpp src/DisplayText.cpp src/Documents.cpp)
target_include_directories(BrowserCheck PRIVATE src vendor)
target_compile_features(BrowserCheck PRIVATE cxx_std_20)
target_compile_definitions(BrowserCheck PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
target_compile_options(BrowserCheck PRIVATE /W4 /EHsc /utf-8)
